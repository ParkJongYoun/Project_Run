if (EditorApplication.isPlaying) throw new System.InvalidOperationException("Capture the authored scene in Edit mode.");
ProjectRun.Race.RaceRunnerView view = null;
foreach (var candidate in UnityEngine.Object.FindObjectsByType<ProjectRun.Race.RaceRunnerView>(FindObjectsInactive.Include, FindObjectsSortMode.None))
    if (candidate.RunnerId == 2) view = candidate;
if (view == null) throw new System.InvalidOperationException("Fox runner missing.");
var stage = new GameObject("temporary_fox_snapshot");
stage.hideFlags = HideFlags.HideAndDontSave;
var meshes = new System.Collections.Generic.List<Mesh>();
Camera camera = null;
RenderTexture rt = null;
Texture2D texture = null;
var previous = RenderTexture.active;
bool asyncShaders = ShaderUtil.allowAsyncCompilation;
bool sceneWasDirty = view.gameObject.scene.isDirty;
Bounds bounds = new Bounds();
bool first = true;
try
{
    foreach (var skin in view.GetComponentsInChildren<SkinnedMeshRenderer>())
    {
        if (!skin.enabled) continue;
        var mesh = new Mesh(); skin.BakeMesh(mesh); meshes.Add(mesh);
        var part = new GameObject(skin.name); part.layer = 31;
        part.transform.SetParent(stage.transform, false);
        part.transform.SetPositionAndRotation(skin.transform.position, skin.transform.rotation);
        part.transform.localScale = Vector3.one;
        part.AddComponent<MeshFilter>().sharedMesh = mesh;
        var renderer = part.AddComponent<MeshRenderer>(); renderer.sharedMaterials = skin.sharedMaterials;
        renderer.renderingLayerMask = 3;
        if (first) { bounds = renderer.bounds; first = false; } else bounds.Encapsulate(renderer.bounds);
    }
    camera = new GameObject("temporary_snapshot_camera").AddComponent<Camera>();
    camera.gameObject.hideFlags = HideFlags.HideAndDontSave;
    camera.enabled = false; camera.cullingMask = 1 << 31;
    camera.clearFlags = CameraClearFlags.SolidColor;
    camera.backgroundColor = new Color(.045f, .075f, .14f, 1);
    camera.fieldOfView = 38; camera.aspect = 1;
    var direction = view.transform.TransformDirection(new Vector3(.95f, .35f, 1.45f)).normalized;
    var rotation = Quaternion.LookRotation(-direction, Vector3.up);
    float distance = 0;
    foreach (var filter in stage.GetComponentsInChildren<MeshFilter>())
    foreach (var vertex in filter.sharedMesh.vertices)
    {
        var projected = Quaternion.Inverse(rotation) * (filter.transform.TransformPoint(vertex) - bounds.center);
        distance = Mathf.Max(distance, Mathf.Max(Mathf.Abs(projected.x),Mathf.Abs(projected.y))/Mathf.Tan(19*Mathf.Deg2Rad)-projected.z);
    }
    camera.transform.position = bounds.center + direction * distance * 1.14f;
    camera.transform.rotation = rotation;
    rt = RenderTexture.GetTemporary(1024,1024,24,RenderTextureFormat.ARGB32);
    camera.targetTexture = rt;
    ShaderUtil.allowAsyncCompilation = false;
    camera.Render();
    RenderTexture.active = rt;
    texture = new Texture2D(1024,1024,TextureFormat.RGB24,false);
    texture.ReadPixels(new Rect(0,0,1024,1024),0,0); texture.Apply();
    string path = "D:/Project_Run/Run/DataBundles/영상/여우_필살기/여우_필살기_입력이미지_1024.png";
    System.IO.File.WriteAllBytes(path,texture.EncodeToPNG());
    return "Saved "+path+"; source="+view.name+"; bounds="+bounds+"; scene was dirty="+sceneWasDirty+"; pose and scene not saved or modified.";
}
finally
{
    ShaderUtil.allowAsyncCompilation = asyncShaders;
    RenderTexture.active = previous;
    if(camera!=null) {camera.targetTexture=null;UnityEngine.Object.DestroyImmediate(camera.gameObject);}
    if(rt!=null) RenderTexture.ReleaseTemporary(rt);
    if(texture!=null) UnityEngine.Object.DestroyImmediate(texture);
    UnityEngine.Object.DestroyImmediate(stage);
    foreach(var mesh in meshes) UnityEngine.Object.DestroyImmediate(mesh);
}
