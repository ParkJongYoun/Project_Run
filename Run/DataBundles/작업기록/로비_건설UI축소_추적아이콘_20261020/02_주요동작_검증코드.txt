var p=UnityEngine.Object.FindFirstObjectByType<ProjectRun.Ranch.Sys_RanchPlacement>();
var c=p.WorldCamera.GetComponent<ProjectRun.Ranch.Sys_RanchCamera>();
var ui=UnityEngine.Object.FindFirstObjectByType<ProjectRun.Race.UIPanel_Ranch>();
var flags=System.Reflection.BindingFlags.Instance|System.Reflection.BindingFlags.NonPublic;
var checks=new System.Collections.Generic.List<string>();
Action<bool,string> check=(ok,name)=>{if(!ok)throw new Exception(name);checks.Add(name);};
Action follow=()=>ui.GetType().GetMethod("UpdateConstructionView",flags).Invoke(ui,null);
Action<string,object[]> pointer=(name,args)=>p.GetType().GetMethod(name,flags).Invoke(p,args);
Action<UnityEngine.UI.Button> click=button=>UnityEngine.EventSystems.ExecuteEvents.Execute(button.gameObject,new UnityEngine.EventSystems.PointerEventData(UnityEngine.EventSystems.EventSystem.current){button=UnityEngine.EventSystems.PointerEventData.InputButton.Left},UnityEngine.EventSystems.ExecuteEvents.pointerClickHandler);
p.SetBuildMode(false);c.ResetView();c.ZoomOut();
c.GetType().GetMethod("ApplyYaw",flags).Invoke(c,new object[]{12f});
var savedPosition=c.transform.position;var savedRotation=c.transform.rotation;float savedSize=c.WorldCamera.orthographicSize;
p.SetBuildMode(true);follow();
check(ui.Drawer.rect.height==120,"Compact construction dock is 120 reference pixels tall");
check(ui.Actions.GetComponentsInChildren<UnityEngine.UI.Text>(true).Length==0,"Floating toolbar has no placement text");
check(ui.Actions.GetComponentsInChildren<UnityEngine.UI.Image>(true).Count(x=>x.name=="img_symbol"&&x.sprite!=null)==4,"All four generated icons are connected");
var buttons=ui.Actions.GetComponentsInChildren<UnityEngine.UI.Button>(true);
check(buttons.All(x=>((RectTransform)x.transform).rect.width==36),"Action buttons are reduced to 36 pixels wide");
check(Mathf.Abs(((RectTransform)buttons[1].transform).anchoredPosition.x-((RectTransform)buttons[0].transform).anchoredPosition.x)==40,"Buttons use a compact four-pixel gap");
check(c.WorldCamera.orthographicSize>13.91f&&c.WorldCamera.orthographicSize<16,"Zoom is slightly farther than the first compact view");
check(Vector3.Dot(c.transform.forward,Vector3.down)>.9999f,"Construction remains exactly top down");
var corners=new Vector3[4];ui.Drawer.GetWorldCorners(corners);float dockTop=corners[1].y;
var gridBottom=c.WorldCamera.WorldToScreenPoint(p.GridOrigin);var gridTop=c.WorldCamera.WorldToScreenPoint(p.GridOrigin+new Vector3(p.GridSize.x,0,p.GridSize.y)*p.CellSize);
check(gridBottom.y>dockTop&&gridTop.y<Screen.height&&gridBottom.x>0&&gridTop.x<Screen.width,"Entire farm remains visible above compact dock");
p.BeginNew("feed_store");p.MovePreview(12,1);follow();
var anchor=ui.Actions.anchoredPosition;p.MovePreview(13,3);follow();
check(Vector2.Distance(anchor,ui.Actions.anchoredPosition)>10,"Toolbar follows the selected facility");
var position=c.transform.position;var rotation=c.transform.rotation;float size=c.WorldCamera.orthographicSize;
var from=c.WorldCamera.WorldToScreenPoint(p.Position(p.Preview));
pointer("BeginConstructionPointer",new object[]{(Vector2)from,false,0});
var candidate=new ProjectRun.Ranch.Data_RanchPlacement{FacilityId="feed_store",X=13,Y=0};
pointer("MoveConstructionPointer",new object[]{(Vector2)c.WorldCamera.WorldToScreenPoint(p.Position(candidate))});follow();
check(p.IsDraggingPreview&&!ui.Actions.GetComponent<CanvasGroup>().blocksRaycasts,"Toolbar does not intercept an active facility drag");
check(c.transform.position==position&&c.transform.rotation==rotation&&c.WorldCamera.orthographicSize==size,"Facility drag keeps construction camera fixed");
pointer("ClearPointer",new object[0]);follow();
check(ui.Actions.GetComponent<CanvasGroup>().blocksRaycasts,"Toolbar becomes clickable after drag release");
int count=p.Layout.Facilities.Count;string placedId=p.Preview.InstanceId;
check(ui.ConfirmButton.interactable,"Check button is enabled at a valid position");
click(ui.ConfirmButton);check(p.Layout.Facilities.Count==count+1&&p.Preview==null,"Check icon confirms exactly one facility");
p.BeginMove(placedId);follow();check(ui.RemoveButton.interactable,"Storage icon is enabled for existing facilities");
click(ui.RemoveButton);check(p.Layout.Facilities.Count==count&&p.Preview==null,"Storage icon recovers the selected facility");
p.BeginNew("feed_store");var blocked=p.Layout.Facilities.First(x=>x.FacilityId=="shelter");p.DragPreviewToScreen(c.WorldCamera.WorldToScreenPoint(p.Position(blocked)));follow();
check(!ui.ConfirmButton.interactable&&p.FootprintMarker.GetComponent<Renderer>().sharedMaterial==p.InvalidMaterial,"Overlap is indicated by red footprint and disabled check");
click(ui.ConfirmButton);check(p.Layout.Facilities.Count==count&&p.Preview!=null,"Disabled check cannot confirm an overlap");
var rotate=buttons.First(x=>x.name=="btn_rotate");int oldRotation=p.Preview.Rotation;click(rotate);
check(p.Preview.Rotation==(oldRotation+1)%4,"Curved-arrow icon rotates the preview");
click(buttons.First(x=>x.name=="btn_cancel"));check(p.Preview==null&&!ui.Actions.gameObject.activeSelf,"X icon cancels and hides the toolbar");
p.BeginNew("training");p.DragPreviewToScreen(new Vector2(Screen.width-10,ui.Drawer.position.y+200));
check(p.Preview.X>=0&&p.Preview.Y>=0&&p.Preview.X+p.Catalog.Find(p.Preview.FacilityId).Size(p.Preview.Rotation).x<=p.GridSize.x,"Edge drag keeps the facility inside the visible farm");
p.SetBuildMode(false);
check(c.transform.position==savedPosition&&c.transform.rotation==savedRotation&&Mathf.Abs(c.WorldCamera.orthographicSize-savedSize)<.0001f,"Closing construction restores the exact previous camera view");
check(JsonUtility.ToJson(p.Layout)==JsonUtility.ToJson(p.InitialLayout),"Checks leave the initial farm layout intact");
return new{checks=checks,zoom=size,facilityTypes=p.Catalog.Facilities.Length};
