var p=UnityEngine.Object.FindFirstObjectByType<ProjectRun.Ranch.Sys_RanchPlacement>();
var ui=UnityEngine.Object.FindFirstObjectByType<ProjectRun.Race.UIPanel_Ranch>();
var flags=System.Reflection.BindingFlags.Instance|System.Reflection.BindingFlags.NonPublic;
var follow=ui.GetType().GetMethod("UpdateConstructionView",flags);
var issues=new System.Collections.Generic.List<object>(); float minGap=999;
foreach(var definition in p.Catalog.Facilities) {
 p.BeginNew(definition.Id);
 for(int rotation=0;rotation<4;rotation++) {
  var size=definition.Size(p.Preview.Rotation);
  foreach(var cell in new[]{new Vector2Int(0,0),new Vector2Int(p.GridSize.x-size.x,0),new Vector2Int(8,6)}) {
   p.MovePreview(cell.x,cell.y); follow.Invoke(ui,null);
   Bounds b; if(!p.TryGetPreviewBounds(out b)) {issues.Add(new{facility=definition.Id,error="No renderer bounds"});continue;}
   float modelBottom=float.PositiveInfinity;
   for(int i=0;i<8;i++){var corner=b.center+Vector3.Scale(b.extents,new Vector3((i&1)==0?-1:1,(i&2)==0?-1:1,(i&4)==0?-1:1));modelBottom=Mathf.Min(modelBottom,p.WorldCamera.WorldToScreenPoint(corner).y);}
   var corners=new Vector3[4]; ui.Actions.GetWorldCorners(corners);
   float gap=modelBottom-RectTransformUtility.WorldToScreenPoint(null,corners[1]).y;minGap=Mathf.Min(minGap,gap);
   if(gap<0)issues.Add(new{facility=definition.Id,rotation=p.Preview.Rotation,x=cell.x,y=cell.y,gap=gap,valid=p.CanConfirm});
  }
  p.Rotate();
 }
}
p.Cancel(); return new{size=p.WorldCamera.orthographicSize,minGap=minGap,issues=issues};
