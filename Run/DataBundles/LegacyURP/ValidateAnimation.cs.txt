var pairs = new string[,] {
    {"Assets/ArtBundles/LegacyURP/Mounts/Prefabs/M_horse_001_pawn.prefab", "Assets/OldAsset/Mounts/horse_001/animation/horse_001@horse_run_01.FBX"},
    {"Assets/ArtBundles/LegacyURP/Mounts/Prefabs/M_vehicle_pecopeco_001_pawn.prefab", "Assets/OldAsset/Mounts/vehicle_pecopeco_001/animation/pecopeco_001@vehicle_pecopeco_run_01.FBX"},
    {"Assets/ArtBundles/LegacyURP/Mounts/Prefabs/M_vehicle_wolf_001_pawn.prefab", "Assets/OldAsset/Mounts/vehicle_wolf_001/animation/vehicle_wolf_001@vehicle_wolf_001_run.FBX"},
    {"Assets/ArtBundles/LegacyURP/Mounts/Prefabs/M_rides_alpaca01_pawn.prefab", "Assets/OldAsset/Mounts/rides_alpaca/animation/rides_alpaca01@rides_alpaca01_run.FBX"},
    {"Assets/ArtBundles/LegacyURP/Characters/Prefabs/Players/prefabs/M_character_m_pawn.prefab", "Assets/OldAsset/Characters/Players/animation/m/m@default_run_m.FBX"},
    {"Assets/ArtBundles/LegacyURP/Characters/Prefabs/Players/prefabs/M_character_w_pawn.prefab", "Assets/OldAsset/Characters/Players/animation/w/w@default_run_w.FBX"}
};
var output = new System.Text.StringBuilder("prefab\tclip\tchanged_vertices\tvertex_count\tmax_displacement\tinvalid_vertices\n");
int failed = 0;
for(int row=0;row<pairs.GetLength(0);row++)
{
    var preview = new UnityEditor.PreviewRenderUtility();
    var mesh = new UnityEngine.Mesh();
    try
    {
        var asset = UnityEditor.AssetDatabase.LoadAssetAtPath<UnityEngine.GameObject>(pairs[row,0]);
        var instance = preview.InstantiatePrefabInScene(asset);
        var animator = instance.GetComponentInChildren<UnityEngine.Animator>(true);
        var actor = animator == null ? instance.transform.GetChild(0).gameObject : animator.gameObject;
        foreach(var animation in instance.GetComponentsInChildren<UnityEngine.Animator>(true)) animation.enabled=false;
        var clip = System.Linq.Enumerable.First(System.Linq.Enumerable.OfType<UnityEngine.AnimationClip>(UnityEditor.AssetDatabase.LoadAllAssetsAtPath(pairs[row,1])), c=>!c.name.StartsWith("__preview__"));
        var skin = System.Linq.Enumerable.First(instance.GetComponentsInChildren<UnityEngine.SkinnedMeshRenderer>(true),r=>r.enabled && r.sharedMesh!=null && r.sharedMesh.vertexCount>20);
        clip.SampleAnimation(actor,0);
        skin.BakeMesh(mesh);
        var before=mesh.vertices;
        clip.SampleAnimation(actor,clip.length*0.37f);
        skin.BakeMesh(mesh);
        var after=mesh.vertices;
        int changed=0,invalid=0;float maximum=0;
        for(int i=0;i<after.Length;i++)
        {
            float delta=(before[i]-after[i]).magnitude;
            if(float.IsNaN(delta)||float.IsInfinity(delta))invalid++;
            else{if(delta>0.00001f)changed++;maximum=UnityEngine.Mathf.Max(maximum,delta);}
        }
        if(changed==0 || invalid>0)failed++;
        output.AppendLine(pairs[row,0]+"\t"+clip.name+"\t"+changed+"\t"+after.Length+"\t"+maximum.ToString(System.Globalization.CultureInfo.InvariantCulture)+"\t"+invalid);
    }
    finally{preview.Cleanup();UnityEngine.Object.DestroyImmediate(mesh);}
}
System.IO.File.WriteAllText(System.IO.Path.Combine(ProjectRun.Editor.LegacyUrpImporter.ReportDirectory,"AnimationValidation.tsv"),output.ToString());
return "Failed samples="+failed+"\n"+output.ToString();
